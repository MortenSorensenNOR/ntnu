\section*{Oppgave 1.}
\begin{enumerate}
    \item Rundt -155 dBm
    \item SNR blir $-41.6 - (-155) = 113.4$ dB
    \item RBW er 100 Hz, så støytettheten blir
        \[
            N_0 = -155 - 10 \log_{10}(100) = -155 - 20 = -175 \text{ dBm/Hz}
        \]
    \item Ved RBW lik 100 kHz bir støygulvet $-175 + 10 \log_{10}(10^{5}) = -175 + 50 = -125$ dBm
\end{enumerate}
